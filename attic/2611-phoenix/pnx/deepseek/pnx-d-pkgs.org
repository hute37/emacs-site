;; ---( Python LSP backend switching )----------------------------------------

(defvar-local h7/python-use-eglot nil
  "Non-nil when the current Python buffer uses Eglot instead of lsp-mode.")

(defun h7/python-lsp-switch (&optional backend)
  "Switch the current Python buffer between LSP backends.
BACKEND is `eglot', `lsp', or nil (toggle)."
  (interactive)
  (unless (derived-mode-p 'python-mode 'python-ts-mode)
    (user-error "This command is intended for Python buffers"))
  (let ((target (or backend
                    (if (bound-and-true-p h7/python-use-eglot) 'lsp 'eglot))))
    (pcase target
      ('eglot
       (when (bound-and-true-p lsp-mode)
         (lsp-disconnect)
         (lsp-mode -1))
       (setq-local h7/python-use-eglot t)
       (when (fboundp #'ruff-format-on-save-mode)
         (ruff-format-on-save-mode -1))
       (when (derived-mode-p 'python-ts-mode)
         (setq-local treesit-font-lock-level 3
                     jit-lock-defer-time 0.05
                     jit-lock-chunk-size 1000))
       (font-lock-mode 1)
       (eglot-ensure)
       (message "Python backend: Eglot"))
      ('lsp
       (when (and (fboundp #'eglot-shutdown) (eglot-managed-p))
         (eglot-shutdown))
       (setq-local h7/python-use-eglot nil)
       (when (derived-mode-p 'python-ts-mode)
         (setq-local treesit-font-lock-level 4
                     jit-lock-defer-time nil
                     jit-lock-chunk-size nil))
       (when (fboundp #'ruff-format-on-save-mode)
         (ruff-format-on-save-mode 1))
       (lsp-deferred)
       (message "Python backend: lsp-mode"))
      (_ (user-error "Unknown Python backend: %S" target)))))

(defun h7/python-use-eglot ()
  (interactive) (h7/python-lsp-switch 'eglot))
(defun h7/python-use-lsp ()
  (interactive) (h7/python-lsp-switch 'lsp))
(defalias 'h7/eglot-mode-switch #'h7/python-use-eglot)

(with-eval-after-load 'python
  (define-key python-base-mode-map (kbd "C-c l e") #'h7/python-use-eglot)
  (define-key python-base-mode-map (kbd "C-c l l") #'h7/python-use-lsp))

;;; #PNX(DUP-02/ENH-01): deterministic per-buffer dispatch - exactly one client.

(defun h7/python-notebook-p ()
  "Return non-nil when the current buffer is a Jupytext notebook.
A notebook is either inside a `notebooks' directory or contains a
`# %%' cell marker near the beginning."
  (and buffer-file-name
       (or (string-match-p "/notebooks/" buffer-file-name)
           (save-excursion
             (goto-char (point-min))
             (re-search-forward "^# %%" (min (point-max) 2000) t)))))

(defun h7/python-apply-lsp-backend ()
  "Start exactly one LSP client for the current Python buffer.
Eglot for notebooks, lsp-mode otherwise.  Idempotent."
  (when (derived-mode-p 'python-mode 'python-ts-mode)
    (if (h7/python-notebook-p)
        (h7/python-use-eglot)
      (h7/python-use-lsp))))

(add-hook 'python-mode-hook #'h7/python-apply-lsp-backend)
(add-hook 'python-ts-mode-hook #'h7/python-apply-lsp-backend)
